function pvMergedSpokenIntervals(){
  const join=PV.cfg.instrumental.spokenJoin ?? .75;
  const arr=S.words
    .filter(w=>w.spoken && w.start_time!==null)
    .map(w=>[w.start_time, w.end_time??(w.start_time+.05)])
    .sort((a,b)=>a[0]-b[0]);
  const out=[];
  for(const it of arr){
    if(!out.length || it[0]-out[out.length-1][1] > join) out.push(it.slice());
    else out[out.length-1][1]=Math.max(out[out.length-1][1],it[1]);
  }
  return out;
}
function pvInstrumentalState(t){
  const c=PV.cfg.instrumental;
  const sung=S.words.filter(w=>!w.spoken && w.start_time!==null).sort((a,b)=>a.start_time-b.start_time);
  if(!sung.length) return null;
  const spoken=pvMergedSpokenIntervals();
  for(let i=0;i<sung.length;i++){
    const next=sung[i], prev=i?sung[i-1]:null;
    const base=prev?(prev.end_time??prev.start_time):Math.max(0,S.cfg.introDuration+.25);
    const gap=next.start_time-base;
    const overlaps=spoken
      .map(([a,b])=>[Math.max(a,base),Math.min(b,next.start_time)])
      .filter(([a,b])=>b>a);
    const hasSpoken=overlaps.length>0;
    const longSpoken=overlaps.some(([a,b])=>(b-a)>=(c.spokenMin??4));
    const regularGap=gap>=c.minGap && !hasSpoken;
    if(!longSpoken && !regularGap) continue;

    // R28: un bloque HABLADO largo se trata visualmente como pausa de canto,
    // pero la próxima letra se libera 3 s antes del siguiente START cantado.
    const lead=longSpoken?(c.spokenLead??3):c.lead;
    const hideAt=next.start_time-lead;
    if(t<base+.2 || t>=hideAt) continue;

    const remain=next.start_time-t;
    const countdownRemain=hideAt-t;
    const halfStep=c.span/Math.max(1,c.dots*2);
    const phase=countdownRemain>=c.span?0:Math.min(c.dots*2,Math.max(0,Math.floor((c.span-countdownRemain)/halfStep+.0001)));
    return {remain,phase,spoken:longSpoken};
  }
  return null;
}
